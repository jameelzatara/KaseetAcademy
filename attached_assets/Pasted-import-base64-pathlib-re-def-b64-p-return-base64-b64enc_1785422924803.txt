reels_js = r'''
(function(){
  var track = document.getElementById('reelTrack');
  if(!track) return;
  var cards = [].slice.call(track.querySelectorAll('.reel-card'));
  
  function getCenterIndex(){
    var tr = track.getBoundingClientRect();
    var cx = tr.left + tr.width / 2;
    var bestIdx = 0, minDiff = 1e9;
    cards.forEach(function(c, i){
      var r = c.getBoundingClientRect();
      var diff = Math.abs((r.left + r.width / 2) - cx);
      if(diff < minDiff){ minDiff = diff; bestIdx = i; }
    });
    return bestIdx;
  }
  
  function updateState(){
    var center = getCenterIndex();
    cards.forEach(function(c, i){
      c.classList.toggle('is-center', i === center);
    });
  }
  
  var raf;
  track.addEventListener('scroll', function(){
    cancelAnimationFrame(raf);
    raf = requestAnimationFrame(updateState);
  }, {passive: true});
  
  window.addEventListener('resize', updateState);
  
  function scrollToCard(dir){
    var idx = getCenterIndex();
    var target = Math.max(0, Math.min(cards.length - 1, idx + dir));
    cards[target].scrollIntoView({inline: 'center', block: 'nearest', behavior: 'smooth'});
  }
  
  var pBtn = document.getElementById('reelPrev');
  var nBtn = document.getElementById('reelNext');
  if(pBtn) pBtn.addEventListener('click', function(){ scrollToCard(-1); });
  if(nBtn) nBtn.addEventListener('click', function(){ scrollToCard(1); });
  
  // Initial positioning
  setTimeout(function(){
    if(cards.length){
      cards[Math.floor(cards.length / 2)].scrollIntoView({inline: 'center', block: 'nearest'});
    }
    updateState();
  }, 400);

  // Trigger Instagram embed processor
  var checkCount = 0;
  var timer = setInterval(function(){
    checkCount++;
    if(window.instgrm && window.instgrm.Embeds){
      window.instgrm.Embeds.process();
      setTimeout(updateState, 600);
    }
    if(checkCount > 10) clearInterval(timer);
  }, 800);
})();
'''
