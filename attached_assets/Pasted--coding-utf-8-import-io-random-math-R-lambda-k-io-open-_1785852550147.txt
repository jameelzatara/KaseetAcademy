def wave_thumb(seed, n=42, w=130, h=28):
    rnd=random.Random(seed); step=w/float(n); o=[]
    for i in range(n):
        a=(0.30+rnd.random()*0.70)*(h/2.0-1); x=i*step+step/2
        o.append('<line x1="%.1f" y1="%.1f" x2="%.1f" y2="%.1f"/>'%(x,h/2-a,x,h/2+a))
    return ('<svg viewBox="0 0 %d %d" preserveAspectRatio="none" aria-hidden="true">'
            '<g stroke="currentColor" stroke-width="2" stroke-linecap="round">%s</g></svg>'%(w,h,''.join(o)))

def poster_wave(seed, n=54, w=300, h=120):
    rnd=random.Random(seed); step=w/float(n); o=[]
    for i in range(n):
        a=(0.10+rnd.random()*0.90)*(h/2.0-4); x=i*step+step/2
        o.append('<line x1="%.1f" y1="%.1f" x2="%.1f" y2="%.1f"/>'%(x,h/2-a,x,h/2+a))
    return ('<svg class="pw" viewBox="0 0 %d %d" preserveAspectRatio="none" aria-hidden="true">'
            '<g stroke="rgba(255,193,7,.30)" stroke-width="2.6" stroke-linecap="round">%s</g></svg>'%(w,h,''.join(o)))

def spine(seed, tall=False):
    h=58 if tall else 42; rnd=random.Random(seed); steps=9 if tall else 7; pts=[]
    for i in range(steps+1):
        y=h*i/float(steps); x=20+(0 if i in (0,steps) else (rnd.random()*2-1)*9); pts.append((x,y))
    d='M%.1f,%.1f'%pts[0]
    for i in range(1,len(pts)):
        px,py=pts[i-1]; cx,cy=pts[i]; d+=' Q%.1f,%.1f %.1f,%.1f'%(px,(py+cy)/2,cx,cy)
    return ('        <div class="spine"><svg viewBox="0 0 40 %d" preserveAspectRatio="none" aria-hidden="true">'
            '<path d="%s" fill="none" stroke="url(#sg)" stroke-width="2" stroke-linecap="round"/></svg></div>'%(h,d))

def spectrum(n=34,w=1440,h=420,seed=21):
    rnd=random.Random(seed); step=w/float(n); o=[]
    for i in range(n):
        ht=(0.16+rnd.random()*0.84)*h; x=i*step+step*0.28
        o.append('<rect x="%.1f" y="%.1f" width="%.1f" height="%.1f" rx="3"/>'%(x,h-ht,step*0.44,ht))
    return '<g fill="rgba(255,255,255,.045)">%s</g>'%''.join(o)
